% Each of the manual's nineteen chapters starts a page, and sets the running head. Both are
% done by wrapping `\section` rather than through titlesec: this titlesec has no `\sectionbreak`
% hook, and a `\markboth` placed in its before-code is typeset inside a box, where the output
% routine never sees the mark. Pandoc emits plain `\section{...}` for every heading here.
\let\uredoorigsection\section
\renewcommand{\section}[1]{\clearpage\markboth{#1}{}\uredoorigsection{#1}}

% --- code ---------------------------------------------------------------------
% A left rule beside the shaded block, so a listing is visible as one object even
% where it breaks across a page.
\renewenvironment{Shaded}{%
  \setlength{\topsep}{8pt}%
  \def\FrameCommand{{\color{uredorule}\vrule width 2.5pt}\hspace{0pt}%
    \fboxsep=8pt\colorbox{shadecolor}}%
  \MakeFramed{\advance\hsize-\width\FrameRestore}}%
  {\endMakeFramed}
% A dozen lines in the manual are wider than any readable measure — the toolchain table's
% longest row is 101 characters — so listings wrap rather than run into the margin, with a
% hook in the gutter marking a continuation.
% The longest listing in the manual is 26 lines and fits on a page with room to spare, so no
% listing is allowed to break across one: a two-line widow of code stranded on the next page
% reads as a separate example, which is worse than a short page.
\fvset{samepage=true,
       fontsize=\small, breaklines=true, breakanywhere=true, breakindent=1.6em,
       breaksymbolleft={\hbox{\color{uredomuted}\footnotesize$\hookrightarrow$}},
       breaksymbolsepleft=3pt}
